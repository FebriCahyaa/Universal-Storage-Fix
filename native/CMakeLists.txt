cmake_minimum_required(VERSION 3.22)
project(universal_storage_fix_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
add_library(usf_core STATIC core/storage_engine.cpp security/path_policy.cpp runtime/runtime_info.cpp)
target_include_directories(usf_core PUBLIC include)
add_executable(usf_native_tests tests/native_tests.cpp)
target_link_libraries(usf_native_tests PRIVATE usf_core)
option(USF_ENABLE_ZYGISK "Build Zygisk adapter with supplied Magisk headers" OFF)
if(USF_ENABLE_ZYGISK)
  if(NOT DEFINED ZYGISK_HEADER_DIR)
    message(FATAL_ERROR "Set ZYGISK_HEADER_DIR to verified Magisk Zygisk headers")
  endif()
  add_library(zygisk_usf SHARED zygisk/zygisk_adapter.cpp)
  target_include_directories(zygisk_usf PRIVATE "${ZYGISK_HEADER_DIR}" include)
  target_link_libraries(zygisk_usf PRIVATE usf_core)
endif()
